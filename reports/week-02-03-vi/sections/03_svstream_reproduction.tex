%=======================================================================
\section{Tái hiện Thực nghiệm SVStream}
\label{sec:svstream_reproduction}
%=======================================================================

Chúng tôi đã tái hiện (reproduce) nghiên cứu thực nghiệm của SVStream~\cite{wang2013svstream} (Hình~3--18) dựa trên phiên bản cài đặt trong River cùng khung benchmark tự xây dựng. Bất kỳ sai khác nào về dữ liệu đầu vào đều phản ánh trực tiếp lên kết quả, do đó phần này trước hết nêu rõ những thiết lập đầu vào nào có thể tuân theo bài báo gốc và những điểm nào không thể, kế đến trình bày cách xử lý và hiệu chuẩn các thông số còn bỏ ngỏ, và cuối cùng so sánh chi tiết kết quả theo từng hình.

%-----------------------------------------------------------------------
\subsection{Dữ liệu và Thiết lập Đầu vào: Những Điểm Tuân theo và Khác biệt so với Bài báo}
\label{subsec:svstream_inputs}
%-----------------------------------------------------------------------

\begin{itemize}[leftmargin=1.4em,itemsep=2pt]
  \item \textbf{Ring-Ball và Smileface-Twomoons không thể khớp hoàn toàn với bài báo} (Hình~3 và mọi hình sử dụng hai tập dữ liệu này). Bộ dữ liệu gốc không được công bố công khai và các hình ảnh trong bài báo không có trục tọa độ, nên việc phỏng đoán tọa độ bằng mắt sẽ không giúp dữ liệu tiệm cận bài báo hơn mà chỉ làm tăng tính tùy biến cảm tính. Chúng tôi giữ nguyên các đặc tính bài báo đã mô tả (kích thước mẫu, số lớp, tỷ lệ nhiễu, thứ tự các pha trôi dạt dữ liệu) và hiệu chuẩn thang đo tọa độ \emph{một lần duy nhất} dựa theo giá trị $q$ tối ưu trong Hình~15 của bài báo (Mục~\ref{subsec:svstream_calibration}). Để khớp hoàn toàn cần phải có dữ liệu gốc từ nhóm tác giả.
  \item \textbf{Phương pháp chuẩn hóa các luồng dữ liệu thực tế không được nêu trong bài báo.} Phương pháp này cũng được lựa chọn thông qua cùng một bước hiệu chuẩn duy nhất và được ghi nhận tường minh.
  \item \textbf{KDD CUP'99 và Forest CoverType được lấy mẫu tiền tố.} Lần chạy này sử dụng 10.000 bản ghi đầu tiên của mỗi luồng trong tất cả các hình (Hình~14--16 và 18), trong khi bài báo chạy toàn bộ luồng với 494.020 và 581.012 bản ghi. Khung benchmark hỗ trợ chạy toàn bộ luồng, nhưng sẽ tiêu tốn hàng chục giờ tính toán. Hình~18 ngoài ra còn giữ nguyên thiết lập trước đó (online running z-score, $q=0.02$ / $0.15$, luồng tổng hợp không co giãn). Do đó, các chỉ số trên luồng dữ liệu thực không thể so sánh trực tiếp theo giá trị tuyệt đối với bài báo.
\end{itemize}

%-----------------------------------------------------------------------
\subsection{Hiệu chuẩn các Yếu tố Đầu vào Chưa Xác định}
\label{subsec:svstream_calibration}
%-----------------------------------------------------------------------

\noindent\textbf{Thang đo tọa độ.} Một hàm nhân Gaussian (Gaussian kernel) với độ rộng $q$ trên dữ liệu được co giãn theo tỷ lệ $s$ hoạt động tương đương với $q s^2$ trên dữ liệu chưa co giãn, do đó một lượt quét duy nhất giá trị $q$ trên luồng chưa co giãn sẽ xác định được $s = \sqrt{q^*/q_{\text{paper}}}$. Ring-Ball đạt đỉnh tại $q^*=8$ so với giá trị $9$ trong bài báo ($s=0.943$); Smileface-Twomoons đạt đỉnh tại $q^*=50$ so với $11$ ($s=2.13$). Do thang đo được chọn bằng cách tối đa hóa chỉ số Rand trung bình tại các tham số mặc định của bài báo, nên các đồ thị luồng tổng hợp ở Hình~15 và chỉ số Rand trung bình ở Hình~5--6 và 11--12 không được coi là bằng chứng độc lập.

\noindent\textbf{Chuẩn hóa dữ liệu.} Tại giá trị $q$ của bài báo, chúng tôi tính toán $q \cdot \max\|\mathbf{x}_i-\mathbf{x}_j\|^2$ trên khối (chunk) đầu tiên cho từng phương án chuẩn hóa ứng viên, vì Mục~4.4.2 của bài báo yêu cầu $q$ phải lớn hơn nhẹ so với $1/\max\|\mathbf{x}_i-\mathbf{x}_j\|^2$, và loại bỏ những phương án chuẩn hóa khiến hầu như mọi giá trị nhân đều bằng $0$ hoặc $1$. Trên KDD CUP'99, chỉ có phương pháp online z-score đưa tích này về gần 1 ($0.99$). Trên CoverType, chỉ có min-max giữ lại được giá trị hàm nhân sử dụng được (với các phương pháp khác, hơn 99\% giá trị hàm nhân xấp xỉ $0$), nhưng tích của nó lên tới $27.6$, do đó hàm nhân vẫn còn quá hẹp: đây là lựa chọn ít tệ nhất thay vì một phương án thực sự thỏa mãn tiêu chí (Mục~\ref{sec:issues}, B4). Việc lựa chọn này hoàn toàn chỉ dựa trên thống kê hàm nhân, không sử dụng chỉ số Rand.

\noindent\textbf{Phân tích.} Chỉ có hiện tượng sụp đổ tại $C=1$ là tái hiện được: hai lớp bị hợp nhất thành một cụm duy nhất tương tự bài báo. Tại giá trị $C=0.25$ của bài báo, chúng tôi hoàn toàn không thu được BSV nào (173 SV, 8 cụm), mặc dù chunk dữ liệu vẫn được phân cụm tốt (RI 0.951). Nguyên nhân xuất phát từ ràng buộc hộp trong Công thức~(5): với $\sum_i \beta_i = 1$ và $\beta_i \le C$, một hình cầu chỉ có thể chứa tối đa $\lfloor 1/C \rfloor$ BSV, tức là 4 BSV tại $C=0.25$, trong khi bài báo lại hiển thị 10 BSV ở đây và 14 BSV tại $C=0.3$ (Hình~8). Do đó, tham số $C$ của tác giả nhiều khả năng đã được co giãn theo một thang khác. Khi cài đặt nguyên bản theo công thức toán học, ràng buộc hộp chỉ thực sự phát huy tác dụng khi $C$ được hạ thấp, ví dụ xuống $0.04$ (thu được 12 BSV).

Các chunk yếu nhất là chunk 1 (0.87), chunk 7 (0.92) và các chunk 65--68 (hạ xuống 0.91), thời điểm hai mắt hội tụ vào nhau. Do đó, sự chênh lệch trước đây trên luồng này chủ yếu bắt nguồn từ thang đo dữ liệu, tức là hàm nhân quá rộng so với tọa độ không gian mẫu.

\takeaway{Chỉ có các tham số đầu vào là có thể tuân theo bài báo một cách tuyệt đối; các luồng dữ liệu tổng hợp chỉ có thể được tái tạo và co giãn lại thang đo, và các luồng dữ liệu thực ở đây là các mẫu lấy 10.000 bản ghi tiền tố. Khi thang đo tọa độ được hiệu chuẩn, Smileface-Twomoons và Ring-Ball tiệm cận rất sát chất lượng phân cụm của bài báo, tuy nhiên cơ chế BSV trong Công thức~(5) không bao giờ kích hoạt tại giá trị $C$ của bài báo, do đó việc phát hiện ngoại lai, nghiên cứu độ nhạy $C$ và kích thước tập tóm tắt vẫn còn khác biệt đáng kể so với bài báo gốc.}
